CREATE TABLE paises (
    id_pais SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL
);

CREATE TABLE categorias (
    id_categoria SERIAL PRIMARY KEY,
    nome VARCHAR(50) NOT NULL
);

CREATE TABLE armas (
    id_arma SERIAL PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    alcance_km INTEGER,
    id_pais INTEGER,
    id_categoria INTEGER,

    CONSTRAINT fk_pais
        FOREIGN KEY (id_pais)
        REFERENCES paises (id_pais),

    CONSTRAINT fk_categoria
        FOREIGN KEY (id_categoria)
        REFERENCES categorias (id_categoria)
);

SELECT 
    a.nome AS arma,
    c.nome AS categoria,
    p.nome AS pais_origem,
    a.alcance_km
FROM armas a
JOIN categorias c ON a.id_categoria = c.id_categoria
JOIN paises p ON a.id_pais = p.id_pais;

SELECT 
    a.nome AS arma,
    c.nome AS categoria,
    p.nome AS pais_origem,
    a.alcance_km
FROM armas a
JOIN categorias c ON a.id_categoria = c.id_categoria
JOIN paises p ON a.id_pais = p.id_pais
ORDER BY a.alcance_km DESC;

SELECT 
    p.nome AS pais,
    COUNT(a.id_arma) AS total_armas
FROM paises p
LEFT JOIN armas a ON p.id_pais = a.id_pais
GROUP BY p.nome;
